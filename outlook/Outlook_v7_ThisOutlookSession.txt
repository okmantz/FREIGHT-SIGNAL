' Freight Signal Outlook Engine v7  -  "safe-send" edition
' Paste this entire module into Outlook VBA > ThisOutlookSession (Alt+F11), save, restart Outlook.
'
' New in v7 (mirrors Freight Signal v16: js/core/sendpolicy.js, carrierreply.js, ledger.js):
'   * PERSISTENT LEDGER (freight-signal-processed.tsv): every e-mail is processed exactly once, even across Outlook restarts, and every
'     auto-send is counted there (per conversation, per day, per hour). v6 forgot everything when Outlook closed.
'   * SEND MODES (set in Freight Signal > System): off = drafts only (default) | shadow = drafts only, but every reply is judged
'     "would I have sent this?" | routine = sends only routine replies | full = routine + rate-confirming replies.
'   * Every gate the app uses: exact lane (score 220+), confirmed target rate, no weak rate, no extra charges (lumper/detention/TONU),
'     short thread, per-conversation / per-day / per-hour caps, duplicate-kind window, work hours, <= 600 characters, <= 2 questions,
'     no attachments, no no-reply / own address. Sends are DEFERRED a few minutes (undo window): delete it from Outbox to cancel.
'   * FS_PauseSending / FS_ResumeSending: one click stops all automatic sending (writes freight-signal-paused.txt).
'   * New reading rules and intents: thanks/out-of-office (ignored), wants details, other loads (truck lead), MC/DOT given, redirect,
'     weak rates ("about 1500", "fifteen hundred") that are drafted but never sent, rate ranges, "is it $1,500?" questions.
'   * Better replies: carrier at/under your last offer = agreed; same rate twice = stalemate (flagged); never counters backwards;
'     every message that needs a person still gets a ready-to-edit DRAFT in your Drafts folder.
'   * The inquiry log gets four more columns (Mode, Why, Fresh, Reply) so the app's Cover Desk can show what happened and why.
'   * FS_SelfTest now runs the app's audited corpus (generated by scripts/gen-selftest.mjs).
'   NOT TESTED IN A LIVE OUTLOOK by its author. Keep the mode on "Drafts only" or "Shadow" until FS_SelfTest passes on your machine
'   and the app's readiness meter says you agree with it on 30+ replies.
'
' v6.1: a carrier e-mail that says "St. Louis" / "Ft. Worth" / "Mt. Vernon" now matches a posting stored as "Saint Louis" / "Fort Worth" /
'       "Mount Vernon" (and the reverse). Before, that inquiry was silently dropped: no draft, no count.
'
' New in v6 (everything else is unchanged from v5):
'   * Carrier e-mails are READ properly. Only what the carrier typed is analysed (the quoted history, signature and your own
'     wording are removed first), so a plain "rate?" can never be mistaken for a quote. The same rules the Freight Signal app uses
'     are loaded from freight-signal-config.tsv.
'   * One right reply per situation: "rate?" -> "How close can you come to $X?" ; "still available?" -> yes + ask for rate and ETA ;
'     a rate -> book / split / counter ladder ; "ok" to your counter -> confirm and ask for ETA ; anything unclear -> no reply, flagged.
'   * A rate is only ever logged (and "booking band" only ever shown) when the carrier actually typed one.
'   * OPTIONAL AUTO-SEND (Freight Signal > System > Danger zone). Two separate switches, both OFF by default:
'       inbound  = replies to carrier e-mails are SENT instead of saved as drafts (only when the app is certain it understood them)
'       outbound = outreach drafts made in the app are SENT instead of saved
'     Guards: a cap of sends per hour, no-reply / own-address recipients are never sent to, every e-mail is processed once.
'   * FS_SelfTest checks the reading rules against 77 sample replies, right inside Outlook.
'
' With both switches off (the default) nothing is ever sent: every reply is a draft, exactly as in v5.
'
' Folder files (all ASCII, tab separated):
'   freight-signal-postings.tsv        written by the app   (read here)
'   freight-signal-config.tsv          written by the app   (rate window, ladder, wording, reading rules, send switches)
'   freight-signal-inquiries.tsv       appended here        (the app imports it into the Inquiry Log)
'   freight-signal-outlook-status.txt  written here         (the app shows "Outlook connected")
'   freight-signal-outbox.tsv          written by the app   (drafts waiting to be created in Outlook)
'   freight-signal-drafts-done.tsv     appended here        (which outbox drafts were created / sent / failed)
' =====================================================================================================

' Create the Outlook drafts that Freight Signal has queued (they also appear on their own when Outlook starts, new mail arrives,
' or you click into Outlook).
' One click stops ALL automatic sending (replies still become drafts). Run FS_ResumeSending to continue.
Public Sub FS_PauseSending()
    On Error Resume Next
    Dim p As String
    p = FeedPath(FILE_PAUSE)
    If Len(p) = 0 Then MsgBox "The Freight Signal folder was not found.", vbExclamation, "Freight Signal Engine": Exit Sub
    AppendLine p, "paused " & Format$(Now, "yyyy-mm-dd hh:nn:ss")
    WriteStatus "Sending paused"
    MsgBox "Automatic sending is PAUSED. Replies are still drafted; nothing leaves by itself. Run FS_ResumeSending to continue.", vbInformation, "Freight Signal Engine"
End Sub

Public Sub FS_ResumeSending()
    On Error Resume Next
    Dim p As String
    p = FeedPath(FILE_PAUSE)
    If Len(p) > 0 Then If Len(Dir$(p)) > 0 Then Kill p
    WriteStatus "Sending resumed"
    MsgBox "Automatic sending is resumed (mode: " & mSendMode & ").", vbInformation, "Freight Signal Engine"
End Sub

' Process everything that is already unread in the Inbox (for e-mails that arrived while Outlook / the macro was off).
Public Sub ProcessUnreadNow()
    On Error GoTo EH
    Dim it As Object, n As Long, items As Outlook.Items
    Set items = Application.GetNamespace("MAPI").GetDefaultFolder(OL_FOLDER_INBOX).Items
    items.Sort "[ReceivedTime]", False
    For Each it In items
        If TypeOf it Is Outlook.MailItem Then
            If it.UnRead Then
                If Not IsBlockedSender(it) And Not IsAutomatedMessage(it) And Not IsForward(it) Then
                    ProcessIncomingCarrierEmail it: n = n + 1
                End If
            End If
        End If
    Next it
    MsgBox "Checked " & n & " unread e-mail(s).", vbInformation, "Freight Signal Engine"
    Exit Sub
EH:
    LogError "ProcessUnreadNow"
End Sub

'============================== INBOUND ==============================
Private Sub InboxItems_ItemAdd(ByVal Item As Object)
    On Error GoTo EH
    If mProcessing Or Item Is Nothing Then Exit Sub
    If Not TypeOf Item Is Outlook.MailItem Then Exit Sub

    Dim mail As Outlook.MailItem
    Set mail = Item
    If Not mail.UnRead Then Exit Sub
    If IsBlockedSender(mail) Or IsAutomatedMessage(mail) Or IsForward(mail) Then Exit Sub

    mProcessing = True
    ProcessIncomingCarrierEmail mail
    ProcessOutbox False
Done:
    mProcessing = False
    Exit Sub
EH:
    LogError "InboxItems_ItemAdd"
    Resume Done
End Sub

'============================== CARRIER E-MAIL ENGINE (v6) ==============================
' Reads a carrier's reply the same way the Freight Signal app does (js/core/carrierreply.js). The patterns come from
' freight-signal-config.tsv ("rule" lines), so the app and this macro can never disagree about what an e-mail means.
' If the rules are missing from the config, NOTHING is drafted or sent: the e-mail is flagged for you instead.
Private Function RulesReady() As Boolean
    RulesReady = False
    If mRules Is Nothing Then Exit Function
    RulesReady = (mRules.Exists("ratedollar") And mRules.Exists("asksrate") And mRules.Exists("cutline") And mRules.Exists("ownphrase") And mRules.Exists("accepts") And mRules.Exists("ratebare") And mRules.Exists("weaknum") And mRules.Exists("noiseid") And mRules.Exists("ack") And mRules.Exists("ooo") And mRules.Exists("wantsdetails") And mRules.Exists("extras"))
End Function

Private Sub AddRule(ByVal name As String, ByVal idx As Long, ByVal src As String)
    Dim c As Collection
    name = LCase$(Trim$(name))
    If mRules Is Nothing Then Set mRules = CreateObject("Scripting.Dictionary")
    If Not mRules.Exists(name) Then Set c = New Collection: mRules.Add name, c
    mRules(name).Add src
End Sub

Private Function Rx(ByVal pattern As String, Optional ByVal isGlobal As Boolean = False, Optional ByVal multi As Boolean = True) As Object
    Dim r As Object
    Set r = CreateObject("VBScript.RegExp")
    r.IgnoreCase = True: r.Global = isGlobal: r.Multiline = multi: r.Pattern = pattern
    Set Rx = r
End Function

Private Function Rule1(ByVal name As String) As String
    If Not mRules Is Nothing Then If mRules.Exists(LCase$(name)) Then Rule1 = CStr(mRules(LCase$(name))(1))
End Function

Private Function AnyOf(ByVal name As String, ByVal text As String) As Boolean
    Dim v As Variant
    If mRules Is Nothing Then Exit Function
    If Not mRules.Exists(LCase$(name)) Then Exit Function
    For Each v In mRules(LCase$(name))
        If Rx(CStr(v), False, True).Test(text) Then AnyOf = True: Exit Function
    Next v
End Function

Private Function TrimWs(ByVal s As String) As String
    Do While Len(s) > 0
        If InStr(" " & vbTab & vbLf & vbCr, Left$(s, 1)) > 0 Then s = Mid$(s, 2) Else Exit Do
    Loop
    Do While Len(s) > 0
        If InStr(" " & vbTab & vbLf & vbCr, Right$(s, 1)) > 0 Then s = Left$(s, Len(s) - 1) Else Exit Do
    Loop
    TrimWs = s
End Function

Private Function NormLine(ByVal s As String) As String
    NormLine = Trim$(Rx("[^a-z0-9$]+", True, False).Replace(LCase$(s), " "))
End Function

Private Function IsCutLine(ByVal ln As String) As Boolean
    IsCutLine = AnyOf("cutline", ln)
End Function

' 1. What the carrier actually typed: quoted history, signatures, our own wording and lines copied from our own text are removed.
Private Function FreshText(ByVal body As String, ByVal ownText As String) As String
    Dim lines() As String, ol() As String, i As Long, j As Long, ln As String, outText As String, hdr As Long, n As String
    Dim own As Object, rxOwn As Object, rxSign As Object, rxOn As Object, rxWrote As Object, rxFrom As Object, rxHdr As Object, rxQuote As Object
    body = Replace(Replace(Replace(body, vbCrLf, vbLf), vbCr, vbLf), ChrW(160), " ")
    ownText = Replace(Replace(Replace(ownText, vbCrLf, vbLf), vbCr, vbLf), ChrW(160), " ")
    Set own = CreateObject("Scripting.Dictionary")
    ol = Split(ownText, vbLf)
    For i = 0 To UBound(ol)
        n = NormLine(ol(i))
        If Len(n) >= 6 Then own(n) = True
    Next i
    Set rxOwn = Rx(Rule1("ownphrase"), False, True)
    Set rxSign = Rx(Rule1("signoffline"), False, True)
    Set rxOn = Rx("^\s*on\s.{3,}$", False, True)
    Set rxWrote = Rx("wrote:\s*$", False, True)
    Set rxFrom = Rx("^\s*from:\s", False, True)
    Set rxHdr = Rx("^\s*(?:sent|date|to|subject|cc):", False, True)
    Set rxQuote = Rx("^\s*>", False, True)
    lines = Split(body, vbLf)
    For i = 0 To UBound(lines)
        ln = lines(i)
        If IsCutLine(ln) Then Exit For
        If rxSign.Test(ln) Then
            If Len(TrimWs(outText)) > 0 Then Exit For
            GoTo NextLine
        End If
        If rxOn.Test(ln) And i < UBound(lines) Then
            If rxWrote.Test(lines(i + 1)) Then Exit For
        End If
        If rxFrom.Test(ln) Then
            hdr = 0
            For j = i + 1 To i + 7
                If j <= UBound(lines) Then If rxHdr.Test(lines(j)) Then hdr = hdr + 1
            Next j
            If hdr >= 2 Then Exit For
        End If
        If rxQuote.Test(ln) Then GoTo NextLine
        If rxOwn.Test(ln) Then GoTo NextLine
        If own.Exists(NormLine(ln)) Then GoTo NextLine
        outText = outText & ln & vbLf
NextLine:
    Next i
    outText = Rx("\n{3,}", True, False).Replace(outText, vbLf & vbLf)
    FreshText = TrimWs(outText)
End Function

Private Function ToNum(ByVal s As String) As Double
    ToNum = Val(Replace(s, ",", ""))
End Function

Private Sub AddCand(ByVal v As Double)
    Dim i As Long
    If v < mMinRate Or v > mMaxRate Then Exit Sub
    For i = 0 To mCandN - 1
        If mCand(i) = v Then Exit Sub
    Next i
    If mCandN < 20 Then mCand(mCandN) = v: mCandN = mCandN + 1
End Sub

' blank out phones, ZIPs, MC/DOT/reference numbers, street addresses, dates and times so a stray number is never read as a rate
Private Function StripNoise(ByVal text As String) As String
    Dim k As Variant, t As String
    t = text
    For Each k In Array("phone", "noiseid", "noiseaddress", "noisezip", "noisedate", "noisetime")
        If Len(Rule1(CStr(k))) > 0 Then t = Rx(Rule1(CStr(k)), True, False).Replace(t, " ")
    Next k
    StripNoise = t
End Function

Private Function YearLike(ByVal matched As String, ByVal v As Double) As Boolean
    YearLike = (InStr(matched, "$") = 0 And v >= 2020 And v <= 2035)
End Function

' 2. The rate in the carrier's own words. Sets mCandN / mCand() and the flags; exactly one distinct plausible amount = found.
'    weak = the amount was not rate-shaped ("fifteen hundred", a lone number, "about 1500"): drafted, never sent.
Private Sub ScanRates(ByVal fresh As String, ByRef found As Boolean, ByRef rate As Double, ByRef ambiguous As Boolean, ByRef perMile As Boolean, ByRef weak As Boolean, ByRef approx As Boolean)
    Dim mt As Object, v As Double, ms As Object, n0 As Long, w As String, n As Long, a As Double, b As Double, clean As String
    mCandN = 0: found = False: rate = 0: weak = False: approx = False
    For Each mt In Rx(Rule1("ratedollar"), True, True).Execute(fresh)
        AddCand ToNum(CStr(mt.SubMatches(0)))
    Next mt
    For Each mt In Rx(Rule1("ratekeyword"), True, True).Execute(fresh)
        v = ToNum(CStr(mt.SubMatches(1)))
        If Not YearLike(mt.Value, v) Then AddCand v
    Next mt
    For Each mt In Rx(Rule1("ratesuffix"), True, True).Execute(fresh)
        AddCand ToNum(CStr(mt.SubMatches(1)))
    Next mt
    If Len(Rule1("rateusdafter")) > 0 Then
        For Each mt In Rx(Rule1("rateusdafter"), True, True).Execute(fresh)
            AddCand ToNum(CStr(mt.SubMatches(1)))
        Next mt
        For Each mt In Rx(Rule1("rateusdbefore"), True, True).Execute(fresh)
            AddCand ToNum(CStr(mt.SubMatches(0)))
        Next mt
        For Each mt In Rx(Rule1("rateoknum"), True, True).Execute(fresh)
            AddCand ToNum(CStr(mt.SubMatches(0)))
        Next mt
        For Each mt In Rx(Rule1("rateworks"), True, True).Execute(fresh)
            v = ToNum(CStr(mt.SubMatches(1)))
            If Not YearLike(mt.Value, v) Then AddCand v
        Next mt
    End If
    For Each mt In Rx(Rule1("ratek"), True, True).Execute(fresh)
        v = Val(CStr(mt.SubMatches(1)))
        If v >= 0.25 And v <= 20 Then AddCand Int(v * 1000 + 0.5)
    Next mt
    Set ms = Rx(Rule1("ratebare"), False, False).Execute(fresh)
    If ms.Count > 0 Then AddCand ToNum(CStr(ms(0).SubMatches(0)))
    perMile = Rx(Rule1("permile"), True, True).Test(fresh)
    If Len(Rule1("raterange")) > 0 Then
        For Each mt In Rx(Rule1("raterange"), True, True).Execute(StripNoise(fresh))
            a = ToNum(CStr(mt.SubMatches(0))): b = ToNum(CStr(mt.SubMatches(1)))
            If a >= mMinRate And a <= mMaxRate And b >= mMinRate And b <= mMaxRate And a <> b Then AddCand a: AddCand b
        Next mt
    End If
    ' weak candidates only when nothing rate-shaped was found
    If mCandN = 0 And Not perMile And Len(Rule1("weaknum")) > 0 Then
        For Each mt In Rx(Rule1("ratehundred"), True, True).Execute(fresh)
            w = LCase$(Replace(Replace(CStr(mt.SubMatches(0)), " ", ""), "-", ""))
            If IsNumeric(w) Then
                n = CLng(w)
            Else
                Select Case w
                    Case "ten": n = 10
                    Case "eleven": n = 11
                    Case "twelve": n = 12
                    Case "thirteen": n = 13
                    Case "fourteen": n = 14
                    Case "fifteen": n = 15
                    Case "sixteen": n = 16
                    Case "seventeen": n = 17
                    Case "eighteen": n = 18
                    Case "nineteen": n = 19
                    Case "twenty": n = 20
                    Case "twentyone": n = 21
                    Case "twentytwo": n = 22
                    Case "twentythree": n = 23
                    Case "twentyfour": n = 24
                    Case "twentyfive": n = 25
                    Case Else: n = 0
                End Select
            End If
            If n > 0 Then AddCand n * 100: weak = True
        Next mt
        clean = StripNoise(fresh)
        For Each mt In Rx(Rule1("weaknum"), True, True).Execute(clean)
            v = ToNum(CStr(mt.SubMatches(0)))
            If v >= mMinRate And v <= mMaxRate And Not (v >= 2020 And v <= 2035) Then
                n0 = mCandN: AddCand v
                If mCandN > n0 Then weak = True
            End If
        Next mt
    End If
    ambiguous = (mCandN > 1)
    If mCandN = 1 And Not perMile Then found = True: rate = mCand(0)
    If found And Len(Rule1("approx")) > 0 Then approx = Rx(Rule1("approx"), True, True).Test(fresh)
    weak = found And (weak Or approx)
End Sub

' another amount besides the rate ("1500 but 1400 if I load Friday")
Private Function HasOtherNumber(ByVal text As String, ByVal rate As Double) As Boolean
    Dim t As String, mt As Object, v As Double
    t = StripNoise(text)
    For Each mt In Rx(Rule1("othernum"), True, True).Execute(t)
        v = ToNum(CStr(mt.SubMatches(0)))
        If v <> rate And v >= 100 Then HasOtherNumber = True: Exit Function
    Next mt
End Function

' 3. What the carrier meant. Mirrors classify() / tagsOf() in js/core/carrierreply.js, same order of checks.
Private Sub AnalyseEmail(ByVal body As String, ByVal ownText As String, ByRef a As TAnalysis)
    Dim f As String, one As String, i As Long, hasRateish As Boolean, mm As Object, lines() As String, nz As Long, firstLine As String
    a.Intent = "unclear": a.Rate = 0: a.Found = False: a.Ambiguous = False: a.PerMile = False: a.Echo = False
    a.Stopped = False: a.Decline = False: a.Pushback = False: a.AsksRate = False: a.AsksAvail = False: a.GivesEta = False: a.GivesTruck = False
    a.NeedsInfo = False: a.WantsCall = False: a.Condition = False: a.Accepts = False: a.AcceptsPlus = False: a.CandText = ""
    a.Weak = False: a.Approx = False: a.Extras = False: a.Ack = False: a.Ooo = False: a.Redirect = False: a.WantsDetails = False: a.OtherLoads = False
    a.GivesInfo = False: a.BareDecline = False: a.BareCall = False: a.QuestionRate = False: a.OnlyPhone = False: a.Mc = ""
    f = FreshText(body, ownText)
    a.Fresh = f
    If Len(f) = 0 Then
        a.Intent = "empty"
        a.Echo = Rx(Rule1("ownphrase"), False, True).Test(body)
        ' "Thank you" alone is a sign-off, so nothing is left: that is a thank-you, not an empty e-mail
        lines = Split(Replace(Replace(body, vbCrLf, vbLf), vbCr, vbLf), vbLf)
        nz = 0: firstLine = ""
        For i = 0 To UBound(lines)
            If Len(Trim$(lines(i))) > 0 Then nz = nz + 1: If nz = 1 Then firstLine = Trim$(lines(i))
        Next i
        If nz >= 1 And nz <= 2 And InStr(firstLine, ">") = 0 And Len(Rule1("ack")) > 0 Then
            If Rx(Rule1("ack"), False, False).Test(firstLine) Then a.Intent = "ack"
        End If
        Exit Sub
    End If
    ScanRates f, a.Found, a.Rate, a.Ambiguous, a.PerMile, a.Weak, a.Approx
    For i = 0 To mCandN - 1
        a.CandText = a.CandText & IIf(i > 0, ", ", "") & FormatCurrency(mCand(i), 0)
    Next i
    hasRateish = (a.Found Or a.Ambiguous Or a.PerMile)
    a.Stopped = AnyOf("stop", f): a.Decline = AnyOf("decline", f): a.Pushback = AnyOf("pushback", f)
    a.AsksRate = AnyOf("asksrate", f): a.AsksAvail = AnyOf("asksavail", f): a.GivesEta = AnyOf("giveseta", f): a.GivesTruck = AnyOf("givestruck", f)
    a.NeedsInfo = AnyOf("needsinfo", f): a.WantsCall = AnyOf("wantscall", f): a.Condition = AnyOf("condition", f)
    a.WantsDetails = AnyOf("wantsdetails", f): a.OtherLoads = AnyOf("asksotherloads", f): a.GivesInfo = AnyOf("givesinfo", f)
    a.Redirect = AnyOf("redirect", f): a.Ooo = AnyOf("ooo", f): a.Extras = AnyOf("extras", f)
    one = TrimWs(f)
    If Len(Rule1("baredecline")) > 0 Then a.BareDecline = Rx(Rule1("baredecline"), False, False).Test(one)
    If Len(Rule1("barecall")) > 0 Then a.BareCall = Rx(Rule1("barecall"), False, False).Test(one)
    If Len(Rule1("ack")) > 0 Then a.Ack = Rx(Rule1("ack"), False, False).Test(one) And Len(one) <= 90
    If Len(Rule1("questionrate")) > 0 Then a.QuestionRate = Rx(Rule1("questionrate"), False, False).Test(one)
    If Len(one) > 0 Then
        If Len(Rx("[^a-z0-9]", True, False).Replace(Rx(Rule1("phone"), True, False).Replace(one, " "), "")) = 0 Then a.OnlyPhone = Rx(Rule1("phone"), False, False).Test(one)
    End If
    a.Accepts = Rx(Rule1("accepts"), False, False).Test(one) And Not a.Condition And Len(one) <= 70
    a.AcceptsPlus = (Not a.Accepts) And Rx(Rule1("acceptword"), False, False).Test(one) And Not a.Condition And InStr(one, "?") = 0 And (a.GivesEta Or a.GivesTruck)
    If a.GivesInfo And Len(Rule1("givesinfo")) > 0 Then
        Set mm = Rx(Rule1("givesinfo"), False, False).Execute(f)
        If mm.Count > 0 Then
            If Rx("\d{5,8}", False, False).Test(mm(0).Value) Then a.Mc = Rx("\d{5,8}", False, False).Execute(mm(0).Value)(0).Value
        End If
    End If
    If a.Ooo And Not hasRateish Then
        a.Intent = "ooo"
    ElseIf a.Stopped Then
        a.Intent = "stop"
    ElseIf (a.Decline Or a.BareDecline) And hasRateish Then
        a.Intent = "mixed"
    ElseIf a.Decline Or a.BareDecline Then
        a.Intent = "decline"
    ElseIf a.Ack And Not hasRateish Then
        a.Intent = "ack"
    ElseIf a.Redirect Then
        a.Intent = "redirect"
    ElseIf a.QuestionRate And hasRateish Then
        a.Intent = "rate_unclear"
    ElseIf a.Ambiguous Or a.PerMile Then
        a.Intent = "rate_unclear"
    ElseIf a.Found And a.Condition And HasOtherNumber(f, a.Rate) Then
        a.Intent = "rate_unclear"
    ElseIf a.Found Then
        a.Intent = "rate"
    ElseIf a.Pushback Then
        a.Intent = "pushback"
    ElseIf a.WantsCall Or a.BareCall Or a.OnlyPhone Then
        a.Intent = "wants_call"
    ElseIf a.GivesInfo And Not a.AsksRate And InStr(f, "?") = 0 Then
        a.Intent = "gives_info"
    ElseIf a.NeedsInfo Then
        a.Intent = "needs_info"
    ElseIf a.Accepts Or a.AcceptsPlus Then
        a.Intent = "accepts"
    ElseIf a.WantsDetails Then
        a.Intent = "wants_details"
    ElseIf a.OtherLoads Then
        a.Intent = "asks_other_loads"
    ElseIf a.AsksRate Then
        a.Intent = "asks_rate"
    ElseIf a.AsksAvail Then
        a.Intent = "asks_available"
    ElseIf a.GivesEta Then
        a.Intent = "gives_eta"
    ElseIf a.GivesTruck Then
        a.Intent = "gives_truck"
    End If
End Sub

    p.Kind = "none": p.Text = "": p.Flag = flag: p.Reason = reason: p.StageNext = stageNext: p.AutoSafe = False
End Sub

Private Sub SetSuggest(ByRef p As TPlan, ByVal kind As String, ByVal text As String)
    p.SuggestKind = kind: p.SuggestText = text
End Sub

' 4. What to do about it: one reply (or nothing, and a flag for you). Mirrors planCarrierReply() in js/core/carrierreply.js.
'    carrierRate = the rate the carrier quoted last time in this conversation (0 if none): used to spot a repeated number.
Private Sub PlanReply(ByRef a As TAnalysis, ByVal t As Double, ByVal stage As String, ByVal ladStage As String, ByVal ourRate As Double, ByVal events As Long, ByVal lastCarrierRate As Double, ByRef p As TPlan)
    Dim questions As Boolean, afterBook As Boolean, r As Double, pct As Double, our As Double, lad As String, countered As Boolean, weakNote As String
    p.Kind = "none": p.Text = "": p.Flag = "": p.Reason = "": p.StageNext = "": p.LadderNext = "": p.OurRateNext = 0: p.Rate = 0
    p.LadStage = "": p.AutoSafe = False: p.Ignore = False: p.Intent = a.Intent: p.LadPct = 0
    p.Quiet = False: p.SuggestKind = "": p.SuggestText = "": p.TruckLead = False: p.MarkInactive = False
    stage = UCase$(Trim$(stage)): If Len(stage) = 0 Then stage = "INQUIRY_REPLY_SENT"
    questions = a.NeedsInfo Or a.WantsCall Or a.Pushback
    If a.Echo Then p.Ignore = True: p.Reason = "Only our own wording came back.": Exit Sub
    If a.Intent = "stop" Then SetManual p, "DO NOT CONTACT - CARRIER ASKED TO STOP", "The carrier asked not to be contacted.", "DECLINED": p.MarkInactive = True: Exit Sub
    If a.Intent = "decline" Then SetManual p, "CARRIER DECLINED - NO REPLY", "The carrier is not interested or cannot cover it.", "DECLINED": Exit Sub
    If a.Intent = "ack" Then SetManual p, "", "The carrier only said thanks: nothing to answer.", stage: p.Ignore = True: p.Quiet = True: Exit Sub
    If a.Intent = "ooo" Then SetManual p, "", "Automatic out-of-office reply: ignored.", stage: p.Ignore = True: p.Quiet = True: Exit Sub
    If stage = "DECLINED" Or stage = "ETA_RECEIVED" Then SetManual p, "FOLLOW-UP - MANUAL REVIEW - NO AUTO REPLY", "This conversation is already finished; a new e-mail needs a person.", stage: Exit Sub
    If events >= 8 Then SetManual p, "MANUAL REVIEW - LONG THREAD", "Long thread - a person should take over.", "": Exit Sub
    afterBook = (stage = "AGREED") Or (stage = "RATE_RESPONSE_SENT" And ladStage = "book")
    If afterBook And (a.Intent = "gives_eta" Or (a.Intent = "accepts" And a.GivesEta)) Then SetManual p, "ETA RECEIVED - BOOK IT", "The carrier answered the ETA question: book it.", "ETA_RECEIVED": Exit Sub
    If stage = "AGREED" Then SetManual p, "FOLLOW-UP - MANUAL REVIEW - NO AUTO REPLY", "The rate is already agreed; anything further needs a person.", "AGREED": Exit Sub
    If afterBook And a.Intent <> "rate" Then SetManual p, "FOLLOW-UP - MANUAL REVIEW - NO AUTO REPLY", "The carrier already has a book-it reply; anything further needs a person.", "FOLLOWUP_FLAGGED": Exit Sub

    Select Case a.Intent
        Case "empty": SetManual p, "MANUAL REVIEW - NO RATE DETECTED", "The e-mail has no new text from the carrier (only quoted history or an attachment).", "FOLLOWUP_FLAGGED"
        Case "mixed": SetManual p, "MANUAL REVIEW - CARRIER REPLY NOT UNDERSTOOD", "The carrier both declined and mentioned a rate: read it yourself.", "FOLLOWUP_FLAGGED"
        Case "redirect"
            SetManual p, "MANUAL REVIEW - CARRIER SENT YOU TO SOMEONE ELSE", "The carrier pointed you to a dispatcher or someone else: send your details to them.", "FOLLOWUP_FLAGGED"
            SetSuggest p, "details", FillTpl2(mTplDetails, t, t)
        Case "rate_unclear"
            If a.PerMile Then
                SetManual p, "MANUAL REVIEW - MORE THAN ONE RATE OR PER-MILE RATE", "The carrier quoted a per-mile rate.", "FOLLOWUP_FLAGGED"
            ElseIf a.QuestionRate Then
                SetManual p, "MANUAL REVIEW - MORE THAN ONE RATE OR PER-MILE RATE", "The carrier asked a question with an amount in it (" & a.CandText & ").", "FOLLOWUP_FLAGGED"
            Else
                SetManual p, "MANUAL REVIEW - MORE THAN ONE RATE OR PER-MILE RATE", "The carrier mentioned more than one amount (" & a.CandText & ").", "FOLLOWUP_FLAGGED"
            End If
            SetSuggest p, "confirmRate", mTplConfirmRate
        Case "pushback"
            SetManual p, "MANUAL REVIEW - CARRIER PUSHED BACK ON RATE", "The carrier said the rate does not work.", "FOLLOWUP_FLAGGED"
            SetSuggest p, "pushback", FillTpl2(mTplPushback, t, t)
        Case "wants_call"
            SetManual p, "CARRIER WANTS A CALL", "The carrier asked you to call or text.", "FOLLOWUP_FLAGGED"
            SetSuggest p, "wantsCall", mTplWantsCall
        Case "needs_info"
            SetManual p, "MANUAL REVIEW - CARRIER ASKED A QUESTION", "The carrier asked something only a person should answer.", "FOLLOWUP_FLAGGED"
            SetSuggest p, "details", FillTpl2(mTplDetails, t, t)
        Case "wants_details"
            p.Kind = "details": p.Text = FillTpl2(mTplDetails, t, t): p.Flag = "CARRIER WANTS DETAILS": p.Reason = "The carrier asked for the load details: resend them and ask for rate and ETA."
            p.StageNext = "FOLLOWUP_RECEIVED": p.AutoSafe = (Not questions)
        Case "asks_other_loads"
            SetManual p, "TRUCK LEAD - CARRIER ASKED FOR OTHER LOADS", "The carrier asked for other loads: a truck lead. Capture the truck and offer a match.", "FOLLOWUP_FLAGGED"
            p.TruckLead = True: SetSuggest p, "otherLoads", mTplOtherLoads
        Case "gives_info"
            SetManual p, "CARRIER SENT MC / DOT", "The carrier sent an MC/DOT number" & IIf(Len(a.Mc) > 0, " (" & a.Mc & ")", "") & ". Saved to the carrier.", "FOLLOWUP_RECEIVED"
            SetSuggest p, "generic", mTplGeneric
        Case "accepts"
            our = ourRate
            If our > 0 And stage = "RATE_RESPONSE_SENT" And ladStage <> "book" And a.GivesEta Then
                SetManual p, "RATE AGREED - BOOK IT (ETA GIVEN)", "The carrier agreed to " & FormatCurrency(our, 0) & " and gave an ETA: book it.", "ETA_RECEIVED"
            ElseIf Not (our > 0) Or stage <> "RATE_RESPONSE_SENT" Or ladStage = "book" Then
                SetManual p, "MANUAL REVIEW - CARRIER REPLY NOT UNDERSTOOD", """Yes / ok"" but there is no counter on the table to agree to.", "FOLLOWUP_FLAGGED"
                SetSuggest p, "generic", mTplGeneric
            Else
                p.Kind = "agreed": p.Text = FillTpl2(mTplAgreed, our, t): p.Flag = "RATE AGREED - BOOK IT": p.Reason = "The carrier agreed to " & FormatCurrency(our, 0) & "."
                p.StageNext = "AGREED": p.LadderNext = "book": p.OurRateNext = our: p.AutoSafe = Not a.Extras
            End If
        Case "rate"
            r = a.Rate
            If a.Weak Then weakNote = " The amount was not clearly labelled as a rate" & IIf(a.Approx, " (""about"")", "") & ": check it." Else weakNote = ""
            If Not (t > 0) Then
                SetManual p, "MANUAL REVIEW - NO TARGET RATE", "Carrier rate " & FormatCurrency(r, 0) & " received, but this posting has no target rate to compare it with.", "FOLLOWUP_RECEIVED"
                p.Rate = r
            ElseIf r < t * 0.5 Or r > t * 3 Then
                SetManual p, "MANUAL REVIEW - RATE LOOKS WRONG", "Carrier rate " & FormatCurrency(r, 0) & " is far from the " & FormatCurrency(t, 0) & " target: check it is a real quote.", "FOLLOWUP_RECEIVED"
                p.Rate = r
            Else
                countered = (stage = "RATE_RESPONSE_SENT" And our_gt0(ourRate) And ladStage <> "book")
                If countered And r <= ourRate Then
                    ' the carrier came down to (or under) the number we last offered: that is a yes
                    p.Kind = "agreed": p.Text = FillTpl2(mTplAgreed, r, t): p.Flag = "RATE AGREED - BOOK IT" & IIf(a.Weak, " (CHECK RATE)", "")
                    p.Reason = "Carrier rate " & FormatCurrency(r, 0) & " is at or under your last offer of " & FormatCurrency(ourRate, 0) & "." & weakNote
                    p.StageNext = "AGREED": p.LadderNext = "book": p.OurRateNext = r: p.Rate = r
                    p.AutoSafe = (Not a.Weak) And (Not a.Extras) And (Not questions) And (Not a.Condition)
                    Exit Sub
                End If
                If countered And lastCarrierRate > 0 And r = lastCarrierRate Then
                    SetManual p, "MANUAL REVIEW - CARRIER REPEATED THE SAME RATE", "The carrier repeated " & FormatCurrency(r, 0) & " after your counter of " & FormatCurrency(ourRate, 0) & ". Decide whether to move or let it go.", "FOLLOWUP_FLAGGED"
                    p.Rate = r: SetSuggest p, "holding", mTplHolding
                    Exit Sub
                End If
                lad = LadderStage(r, t): pct = (r - t) / t * 100#
                If lad = "book" Then
                    our = r
                ElseIf lad = "split" Then
                    our = SplitRate(r, t)
                Else
                    our = t
                End If
                ' never go backwards: a counter lower than the one already sent is not a negotiation
                If countered And lad <> "book" And our < ourRate Then
                    SetManual p, "MANUAL REVIEW - CARRIER REPEATED THE SAME RATE", "The carrier is at " & FormatCurrency(r, 0) & ", you already offered " & FormatCurrency(ourRate, 0) & ". Another counter would go backwards.", "FOLLOWUP_FLAGGED"
                    p.Rate = r: SetSuggest p, "holding", mTplHolding
                    Exit Sub
                End If
                p.Kind = "ladder": p.LadStage = lad: p.LadPct = pct: p.Rate = r
                If lad = "book" Then
                    p.Text = FillTemplate(mTplBook, r, t): p.Flag = "RATE RECEIVED - REVIEW / ETA"
                ElseIf lad = "split" Then
                    p.Text = FillTemplate(mTplSplit, r, t): p.Flag = "RATE RECEIVED - SPLIT COUNTER DRAFTED"
                Else
                    p.Text = FillTemplate(mTplTarget, r, t): p.Flag = ""
                End If
                If a.Weak Then p.Flag = "MANUAL REVIEW - RATE NOT CLEARLY STATED"
                If a.Extras Then p.Flag = "MANUAL REVIEW - CARRIER ASKED FOR EXTRA CHARGES"
                p.Reason = "Carrier rate " & FormatCurrency(r, 0) & " vs target " & FormatCurrency(t, 0) & " (" & Format$(pct, "0.0") & "%)." & weakNote
                p.StageNext = "RATE_RESPONSE_SENT": p.LadderNext = lad: p.OurRateNext = our
                p.AutoSafe = (Not a.Weak) And (Not a.Extras) And (Not questions) And (Not a.Condition)
            End If
        Case "asks_rate"
            If Not (t > 0) Then
                SetManual p, "MANUAL REVIEW - NO TARGET RATE", "The carrier asked for the rate but this posting has no target rate yet.", "FOLLOWUP_FLAGGED"
            Else
                p.Kind = "asks_rate": p.Text = IIf(a.AsksAvail, mTplAvailableLead & " ", "") & FillTpl2(mTplAskRate, t, t)
                p.Reason = "The carrier asked what the rate is: answered with your target."
                p.StageNext = "RATE_RESPONSE_SENT": p.LadderNext = "target": p.OurRateNext = t: p.AutoSafe = Not a.Extras
            End If
        Case "asks_available"
            p.Kind = "available": p.Text = mTplAvailable: p.Reason = "The carrier asked whether the load is still available.": p.StageNext = "FOLLOWUP_RECEIVED": p.AutoSafe = True
        Case "gives_eta"
            p.Kind = "need_rate": p.Text = mTplNeedRate: p.Reason = "The carrier gave timing but no rate.": p.StageNext = "FOLLOWUP_RECEIVED": p.AutoSafe = True
        Case "gives_truck"
            p.Kind = "need_rate_eta": p.Text = mTplNeedRateEta: p.Reason = "The carrier described a truck but gave no rate or ETA.": p.StageNext = "FOLLOWUP_RECEIVED": p.AutoSafe = True: p.TruckLead = True
        Case Else
            SetManual p, "MANUAL REVIEW - CARRIER REPLY NOT UNDERSTOOD", "The reply was not clearly a rate, a question about the rate, an ETA or an answer.", "FOLLOWUP_FLAGGED"
            SetSuggest p, "generic", mTplGeneric
    End Select
End Sub

Private Function our_gt0(ByVal v As Double) As Boolean
    our_gt0 = (v > 0)
End Function

Private Function FillTpl2(ByVal tpl As String, ByVal ourRate As Double, ByVal targetRate As Double) As String
    Dim s As String
    s = Replace(tpl, "{ourRate}", FormatCurrency(ourRate, 0), 1, -1, vbTextCompare)
    s = Replace(s, "{target}", FormatCurrency(targetRate, 0), 1, -1, vbTextCompare)
    FillTpl2 = s
End Function

'---------- persistent ledger: every processed e-mail and every auto-send is recorded in freight-signal-processed.tsv ----------
Private Sub LoadLedger()
    On Error GoTo EH
    Dim txt As String, lines() As String, i As Long, c() As String, tm As Date
    Set mSeen = CreateObject("Scripting.Dictionary")
    Set mSentLog = New Collection
    mLedgerLoaded = True
    If Len(BridgeFolder()) = 0 Then Exit Sub
    txt = ReadAll(FeedPath(FILE_LEDGER))
    If Len(txt) = 0 Then Exit Sub
    txt = Replace(txt, vbCrLf, vbLf)
    lines = Split(txt, vbLf)
    For i = 0 To UBound(lines)
        If Len(lines(i)) > 0 And Left$(lines(i), 10) <> "MessageKey" And Left$(lines(i), 1) <> "#" Then
            c = Split(lines(i), vbTab)
            If UBound(c) >= 4 Then
                If LCase$(Trim$(c(4))) = "seen" Then
                    mSeen(c(0)) = True
                ElseIf LCase$(Trim$(c(4))) = "sent" Then
                    If IsDate(c(1)) Then mSentLog.Add Array(LCase$(DecodeField(c(2))), LCase$(Trim$(c(3))), CDate(c(1)))
                End If
            End If
        End If
    Next i
    Exit Sub
EH:
    LogError "LoadLedger"
End Sub

Private Sub LedgerAppend(ByVal key As String, ByVal convId As String, ByVal kind As String, ByVal mode As String)
    On Error GoTo EH
    Dim p As String
    p = FeedPath(FILE_LEDGER)
    If Len(p) = 0 Then Exit Sub
    If Len(Dir$(p)) = 0 Then AppendLine p, LEDGER_HEADER
    AppendLine p, key & vbTab & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbTab & EncodeField(convId) & vbTab & kind & vbTab & mode
    Exit Sub
EH:
    LogError "LedgerAppend"
End Sub

' One stable key per message: the Internet message id when there is one (survives moves between folders), else the EntryID.
Private Function MessageKey(ByVal mail As Outlook.MailItem) As String
    Dim id As String
    On Error Resume Next
    id = mail.PropertyAccessor.GetProperty("http://schemas.microsoft.com/mapi/proptag/0x1035001F")
    If Len(id) = 0 Then id = mail.EntryID
    On Error GoTo 0
    MessageKey = Replace(Replace(id, vbTab, " "), vbLf, " ")
End Function

Private Function SeenMail(ByVal mail As Outlook.MailItem) As Boolean
    Dim id As String
    id = MessageKey(mail)
    If Len(id) = 0 Then Exit Function
    If Not mLedgerLoaded Or mSeen Is Nothing Then LoadLedger
    If mSeen.Exists(id) Then
        SeenMail = True
    Else
        mSeen(id) = True
        LedgerAppend id, "", "", "seen"
    End If
End Function

Private Sub NoteSend(ByVal convId As String, ByVal kind As String)
    If Not mLedgerLoaded Or mSentLog Is Nothing Then LoadLedger
    mSentLog.Add Array(LCase$(convId), LCase$(kind), Now)
    LedgerAppend "sent:" & Format$(Now, "yyyymmddhhnnss"), convId, kind, "sent"
End Sub

Private Function SentCount(ByVal convId As String, ByVal withinMinutes As Long) As Long
    Dim v As Variant
    If Not mLedgerLoaded Or mSentLog Is Nothing Then LoadLedger
    For Each v In mSentLog
        If Len(convId) = 0 Or v(0) = LCase$(convId) Then
            If withinMinutes <= 0 Then
                SentCount = SentCount + 1
            ElseIf DateDiff("n", v(2), Now) < withinMinutes Then
                SentCount = SentCount + 1
            End If
        End If
    Next v
End Function

Private Function SentKindRecently(ByVal convId As String, ByVal kind As String, ByVal withinMinutes As Long) As Boolean
    Dim v As Variant
    If Not mLedgerLoaded Or mSentLog Is Nothing Then LoadLedger
    For Each v In mSentLog
        If v(0) = LCase$(convId) And v(1) = LCase$(kind) Then
            If DateDiff("n", v(2), Now) < withinMinutes Then SentKindRecently = True: Exit Function
        End If
    Next v
End Function

' Guards for outreach drafted in the app (outbox) when the outbound switch is on. Inbound replies use DecideSend instead.
Private Function MaySend(ByVal switchOn As Boolean, ByVal autoSafe As Boolean, ByVal recipientAddr As String, ByVal recipientCount As Long) As Boolean
    If Not switchOn Then Exit Function
    If Not autoSafe Then Exit Function
    If recipientCount <> 1 Then Exit Function
    If Len(recipientAddr) = 0 Then Exit Function
    If LooksLikeNoReply(recipientAddr) Then Exit Function
    If LCase$(recipientAddr) = OwnAddress() Then Exit Function
    If IsPaused() Then Exit Function
    If SentCount("", 60) >= mMaxSends Then Exit Function
    If SentCount("", 1440) >= mMaxPerDay Then Exit Function
    MaySend = True
End Function

Private Function IsPaused() As Boolean
    Dim p As String
    p = FeedPath(FILE_PAUSE)
    If Len(p) = 0 Then Exit Function
    IsPaused = (Len(Dir$(p)) > 0)
End Function

Private Function LiveMode() As Boolean
    LiveMode = (mSendMode = "routine" Or mSendMode = "full")
End Function

Private Function CountAttachments(ByVal mail As Outlook.MailItem) As Long
    Dim i As Long, cid As String
    On Error Resume Next
    For i = 1 To mail.Attachments.Count
        cid = ""
        cid = mail.Attachments(i).PropertyAccessor.GetProperty("http://schemas.microsoft.com/mapi/proptag/0x3712001F")
        If Len(cid) = 0 Then CountAttachments = CountAttachments + 1
    Next i
End Function

Private Sub AddWhy(ByVal reason As String)
    If Len(mWhy) > 0 Then mWhy = mWhy & " | "
    mWhy = mWhy & reason
End Sub

' Mirrors decideSend() in js/core/sendpolicy.js. Returns True when this reply may leave WITHOUT a person.
' would = it passed every gate (what shadow mode counts, even though nothing is sent). mWhy = the plain-words reasons it was held.
Private Function DecideSend(ByVal mail As Outlook.MailItem, ByRef a As TAnalysis, ByRef p As TPlan, ByVal convId As String, ByVal laneScore As Long, _
                            ByVal targetConfirmed As Boolean, ByVal events As Long, ByVal recipientAddr As String, ByVal recipientCount As Long, ByRef would As Boolean, ByRef delayMinutes As Long) As Boolean
    Dim isCounter As Boolean, isCommit As Boolean, routine As Boolean, needTarget As Boolean, q As Long, i As Long, hr As Long
    mWhy = "": would = False: delayMinutes = 0
    If Len(p.Text) = 0 Or p.Kind = "none" Then AddWhy "There is nothing to send for this message."
    If Not p.AutoSafe Then AddWhy "The reply is one you should look at first."
    If recipientCount <> 1 Then AddWhy "The reply would go to more than one recipient."
    If Len(recipientAddr) = 0 Then AddWhy "The sender address could not be read."
    If LooksLikeNoReply(recipientAddr) Then AddWhy "The sender is a no-reply address."
    If LCase$(recipientAddr) = OwnAddress() Then AddWhy "The message came from your own address."
    If a.Weak Then AddWhy "The rate was not clearly stated."
    If a.Extras Then AddWhy "The carrier mentioned extra charges."
    isCounter = (p.Kind = "ladder" And Len(p.LadderNext) > 0 And p.LadderNext <> "book")
    isCommit = (p.Kind = "agreed") Or (p.Kind = "ladder" And p.LadderNext = "book")
    routine = (p.Kind = "asks_rate" Or p.Kind = "available" Or p.Kind = "need_rate" Or p.Kind = "need_rate_eta" Or p.Kind = "details" Or p.Kind = "inquiry" Or isCounter)
    If Not routine And Not isCommit Then AddWhy "This kind of reply is always sent by a person."
    If isCommit And mSendMode <> "full" Then AddWhy "Replies that confirm a rate are only auto-sent in the ""routine + rate-confirming"" mode."
    needTarget = (p.Kind = "asks_rate" Or p.Kind = "ladder" Or p.Kind = "agreed")
    If needTarget And Not targetConfirmed Then AddWhy "The target rate on this load was suggested, not confirmed by you."
    If laneScore < mMinLane Then AddWhy "The e-mail did not match the load exactly (lane score " & laneScore & ")."
    If events >= 6 Then AddWhy "Long conversation: a person should take over."
    If CountAttachments(mail) > 0 Then AddWhy "The carrier attached a file."
    If Len(p.Text) > mMaxChars Then AddWhy "The reply is longer than usual."
    q = Len(p.Text) - Len(Replace(p.Text, "?", ""))
    If q > 2 Then AddWhy "The reply asks more than two questions."
    If SentCount(convId, 0) >= mMaxPerConv Then AddWhy "Already auto-replied " & SentCount(convId, 0) & " times in this conversation."
    If SentKindRecently(convId, p.Kind, mDupMin) Then AddWhy "The same kind of reply was already sent in this conversation a moment ago."
    If SentCount("", 1440) >= mMaxPerDay Then AddWhy "Daily auto-send limit reached."
    If SentCount("", 60) >= mMaxSends Then AddWhy "Hourly auto-send limit reached."
    If mWorkOnly Then
        hr = Hour(Now)
        If hr < mWorkStart Or hr >= mWorkEnd Then AddWhy "Outside your sending hours."
    End If
    would = (Len(mWhy) = 0)
    If would And IsPaused() Then AddWhy "Sending is paused (FS_ResumeSending to continue).": DecideSend = False: Exit Function
    If would And LiveMode() Then
        DecideSend = True: delayMinutes = mDelayMin
    End If
End Function

' Create the reply above the quoted thread. live = send it (deferred by delayMin minutes so you can still cancel it from the Outbox);
' otherwise it is saved as a draft (and shown when displayIt).
Private Function DeliverReply(ByVal mail As Outlook.MailItem, ByVal text As String, ByVal live As Boolean, ByVal delayMin As Long, ByVal displayIt As Boolean, ByRef wasSent As Boolean) As Boolean
    On Error GoTo EH
    Dim reply As Outlook.MailItem
    wasSent = False
    Set reply = mail.Reply
    reply.Body = text & vbCrLf & vbCrLf & reply.Body
    If live And reply.Recipients.Count <> 1 Then live = False
    If live Then
        If delayMin > 0 Then reply.DeferredDeliveryTime = DateAdd("n", delayMin, Now)
        reply.Send
        wasSent = True
    Else
        reply.Save
        If displayIt Then reply.Display
    End If
    DeliverReply = True
    Exit Function
EH:
    LogError "DeliverReply"
End Function

Private Sub ReloadConfigIfChanged()
    Dim p As String, stamp As String
    On Error Resume Next
    p = FeedPath(FILE_CONFIG)
    If Len(p) = 0 Then Exit Sub
    stamp = CStr(FileDateTime(p)) & "|" & CStr(FileLen(p))
    If stamp <> mConfigStamp Then LoadConfig: mConfigStamp = stamp
End Sub

'============================== INBOUND E-MAILS ==============================
Private Sub ProcessIncomingCarrierEmail(ByVal mail As Outlook.MailItem)
    On Error GoTo EH
    Dim convId As String, stage As String, evt As Long, postingId As String, postingRow As Long, ladStage As String, ourRate As Double, lastCarrier As Double
    Dim idx As Long, targetRate As Double, info As Variant, a As TAnalysis, p As TPlan, addr As String
    Dim sent As Boolean, ok As Boolean, actionText As String, nextStage As String, rateText As String
    Dim live As Boolean, would As Boolean, delayMin As Long, modeText As String, confirmed As Boolean, sendText As String

    If Not LoadPostings(False) Then Exit Sub
    ReloadConfigIfChanged
    convId = GetConversationTrackingID(mail)
    If Len(convId) = 0 Then Exit Sub
    If SeenMail(mail) Then Exit Sub

    If Not mConv.Exists(LCase$(convId)) Then
        ProcessCarrierInquiry mail, convId
        Exit Sub
    End If

    info = mConv(LCase$(convId))
    stage = UCase$(CStr(info(0))): evt = CLng(info(1)): postingId = CStr(info(2)): postingRow = CLng(info(3))
    ladStage = CStr(info(4)): ourRate = CDbl(info(5)): lastCarrier = CDbl(info(6))

    idx = FindPostingIndex(postingId)
    If idx = 0 Then
        ' The posting was covered / deleted in the app since the inquiry started: nothing to draft against.
        MarkMailRead mail
        Exit Sub
    End If
    targetRate = SafeNumeric(PF(idx, "targetrate"))
    confirmed = (LCase$(PF(idx, "targetconfirmed")) <> "suggested")

    If Not RulesReady() Then
        SetMailFlag mail, "MANUAL REVIEW - CARRIER REPLY NOT UNDERSTOOD"
        RecordEvent mail, convId, postingId, postingRow, stage, "", targetRate, "Reply rules are not loaded (open Freight Signal so it can write the config): nothing drafted", _
            "MANUAL REVIEW - CARRIER REPLY NOT UNDERSTOOD", evt + 1, "", ourRate, ladStage, False, "review", "Reading rules are missing from the config file.", "", ""
        Exit Sub
    End If

    AnalyseEmail mail.Body, PostingDetails(idx), a
    PlanReply a, targetRate, stage, ladStage, ourRate, evt, lastCarrier, p
    If p.Ignore Then
        If p.Quiet Then
            RecordEvent mail, convId, postingId, postingRow, stage, "", targetRate, p.Reason, "", evt, p.Intent, ourRate, ladStage, False, "ignored", "", a.Fresh, ""
            MarkMailRead mail
        End If
        Exit Sub
    End If

    nextStage = stage
    If Len(p.StageNext) > 0 Then nextStage = p.StageNext
    If p.Rate > 0 Then rateText = Trim$(Str$(p.Rate)) Else rateText = ""
    addr = GetSenderSMTPAddress(mail)

    If p.Kind = "none" Then
        If Len(p.Flag) > 0 Then SetMailFlag mail, p.Flag
        ' a message that needs a person still gets a ready-to-edit draft in your Drafts folder (never sent, never displayed)
        If Len(p.SuggestText) > 0 Then
            sendText = p.SuggestText
            If p.SuggestKind = "details" Then sendText = PostingDetails(idx) & vbCrLf & vbCrLf & p.SuggestText
            DeliverReply mail, sendText, False, 0, False, sent
        End If
        RecordEvent mail, convId, postingId, postingRow, nextStage, rateText, targetRate, p.Reason & IIf(Len(p.SuggestText) > 0, " A suggested reply is waiting in your Drafts.", ""), p.Flag, evt + 1, p.Intent, ourRate, ladStage, False, _
            "review", "", a.Fresh, p.SuggestText
        If p.StageNext = "DECLINED" Or p.StageNext = "ETA_RECEIVED" Then MarkMailRead mail
        Exit Sub
    End If

    live = DecideSend(mail, a, p, convId, 1000, confirmed, evt, addr, 1, would, delayMin)
    sendText = p.Text
    If p.Kind = "details" Then sendText = PostingDetails(idx) & vbCrLf & vbCrLf & p.Text
    ok = DeliverReply(mail, sendText, live, delayMin, False, sent)
    If Not ok Then Exit Sub
    If sent Then NoteSend convId, p.Kind
    If sent Then
        modeText = "auto"
    ElseIf would And mSendMode = "shadow" Then
        modeText = "shadow"
    Else
        modeText = "draft"
    End If
    If Len(p.Flag) > 0 Then SetMailFlag mail, p.Flag
    Select Case p.Kind
        Case "ladder"
            If p.LadStage = "book" Then
                actionText = "Rate " & Format$(Abs(p.LadPct), "0.0") & "% " & IIf(p.LadPct <= 0, "at/under", "over") & " target (booking band) - ETA confirmation " & IIf(sent, "SENT", "drafted")
            ElseIf p.LadStage = "split" Then
                actionText = "Rate " & Format$(p.LadPct, "0.0") & "% over target - split-the-difference counter " & IIf(sent, "SENT", "drafted") & " (" & FormatCurrency(p.OurRateNext, 0) & ")"
            Else
                actionText = "Rate " & Format$(p.LadPct, "0.0") & "% over target - counter at " & FormatCurrency(targetRate, 0) & IIf(sent, " SENT", " drafted")
            End If
        Case "asks_rate": actionText = "Carrier asked for the rate - answered with your target (" & FormatCurrency(targetRate, 0) & ")" & IIf(sent, " - SENT", "")
        Case "agreed": actionText = "Carrier agreed to " & FormatCurrency(p.OurRateNext, 0) & " - ETA requested" & IIf(sent, " - SENT", "")
        Case "available": actionText = "Carrier asked if the load is available - confirmed and asked for a rate" & IIf(sent, " - SENT", "")
        Case "details": actionText = "Carrier asked for the details - resent them and asked for rate and ETA" & IIf(sent, " - SENT", "")
        Case Else: actionText = "Carrier gave no rate - asked for it" & IIf(sent, " - SENT", "")
    End Select
    If p.Kind = "ladder" Or p.Kind = "agreed" Then
        RecordEvent mail, convId, postingId, postingRow, nextStage, rateText, targetRate, actionText, p.Flag, evt + 1, p.Intent, p.OurRateNext, p.LadderNext, sent, modeText, mWhy, a.Fresh, p.Text
    Else
        RecordEvent mail, convId, postingId, postingRow, nextStage, "", targetRate, actionText, p.Flag, evt + 1, p.Intent, IIf(p.OurRateNext > 0, p.OurRateNext, ourRate), IIf(Len(p.LadderNext) > 0, p.LadderNext, ladStage), sent, modeText, mWhy, a.Fresh, p.Text
    End If
    MarkMailRead mail
    Exit Sub
EH:
    LogError "ProcessIncomingCarrierEmail"
End Sub

Private Sub ProcessCarrierInquiry(ByVal mail As Outlook.MailItem, ByVal convId As String)
    On Error GoTo EH
    Dim idx As Long, details As String, a As TAnalysis, p As TPlan, targetRate As Double, askRate As Boolean, askText As String
    Dim sent As Boolean, rulesOk As Boolean, intentText As String, live As Boolean, would As Boolean, delayMin As Long, modeText As String
    Dim needsPerson As Boolean, lane As Long, freshText2 As String
    idx = FindMatchingPosting(mail.Subject, mail.Body)
    If idx = 0 Then Exit Sub
    details = PostingDetails(idx)
    If Len(details) = 0 Then Exit Sub
    targetRate = SafeNumeric(PF(idx, "targetrate"))
    lane = CLng(mLastScore Mod 1000)
    rulesOk = RulesReady()
    intentText = ""
    If rulesOk Then
        AnalyseEmail mail.Body, details, a
        intentText = a.Intent
        freshText2 = a.Fresh
        If a.Echo Then Exit Sub                                ' only our own wording came back (e.g. a test sent to yourself)
        If a.Intent = "stop" Or a.Intent = "decline" Or a.Intent = "ack" Or a.Intent = "ooo" Then Exit Sub
        askRate = (a.Intent = "asks_rate" And targetRate > 0)
        needsPerson = (a.Intent = "wants_call" Or a.Intent = "pushback" Or a.Intent = "needs_info" Or a.Intent = "rate_unclear" Or a.Intent = "redirect" Or a.Intent = "asks_other_loads" Or a.Intent = "mixed")
    End If
    If askRate Then
        askText = FillTpl2(mTplAskRate, targetRate, targetRate) & vbCrLf & "What is your ETA to pickup?"
    Else
        askText = "What is your best all-in rate? ETA to pickup?"
    End If
    ' only an exact city + state match may be sent without you looking, and never when the carrier needs a person
    p.Kind = "inquiry": p.Text = askText: p.LadderNext = "": p.AutoSafe = rulesOk And (lane >= 220) And Not needsPerson
    live = DecideSend(mail, a, p, convId, lane, True, 0, GetSenderSMTPAddress(mail), 1, would, delayMin)
    If Not rulesOk Then live = False: would = False
    If DeliverReply(mail, details & vbCrLf & vbCrLf & askText, live, delayMin, Not live, sent) Then
        If sent Then NoteSend convId, "inquiry"
        If sent Then
            modeText = "auto"
        ElseIf would And mSendMode = "shadow" Then
            modeText = "shadow"
        Else
            modeText = "draft"
        End If
        RecordEvent mail, convId, PF(idx, "postingid"), CLng(SafeNumeric(PF(idx, "row"))), "INQUIRY_REPLY_SENT", "", targetRate, _
            IIf(askRate, "Inquiry reply " & IIf(sent, "SENT", "draft created") & " (asked for the rate: answered with the target)", "Inquiry reply " & IIf(sent, "SENT", "draft created")), "", 1, intentText, _
            IIf(askRate, targetRate, 0), IIf(askRate, "target", ""), sent, modeText, mWhy, freshText2, askText
        MarkMailRead mail
    End If
    Exit Sub
EH:
    LogError "ProcessCarrierInquiry"
End Sub

' ---- Counter ladder (percent of target). Same rules as the Freight Signal app (core/ladder.js).
Private Function LadderStage(ByVal carrierRate As Double, ByVal targetRate As Double) As String
    Dim pct As Double
    If targetRate <= 0 Then LadderStage = "none": Exit Function
    pct = (carrierRate - targetRate) / targetRate * 100#
    If pct <= mBookPct + 0.000000001 Then
        LadderStage = "book"
    ElseIf pct <= mSplitPct + 0.000000001 Then
        LadderStage = "split"
    Else
        LadderStage = "target"
    End If
End Function

Private Function SplitRate(ByVal carrierRate As Double, ByVal targetRate As Double) As Double
    Dim mid As Double
    mid = (carrierRate + targetRate) / 2#
    If mRound > 0 Then mid = Int(mid / mRound + 0.5) * mRound Else mid = Int(mid + 0.5)
    If mid < targetRate Then mid = targetRate
    If mid > carrierRate Then mid = carrierRate
    SplitRate = mid
End Function

Private Function FillTemplate(ByVal tpl As String, ByVal carrierRate As Double, ByVal targetRate As Double) As String
    Dim s As String
    s = tpl
    s = Replace(s, "{rate}", FormatCurrency(carrierRate, 0), 1, -1, vbTextCompare)
    s = Replace(s, "{target}", FormatCurrency(targetRate, 0), 1, -1, vbTextCompare)
    s = Replace(s, "{split}", FormatCurrency(SplitRate(carrierRate, targetRate), 0), 1, -1, vbTextCompare)
    If targetRate > 0 Then s = Replace(s, "{pct}", Format$(Abs((carrierRate - targetRate) / targetRate * 100#), "0.0") & "%", 1, -1, vbTextCompare)
    FillTemplate = s
End Function

'============================== FOLDER / FILE I/O ==============================
Private Function BridgeFolder() As String
    If Len(mFolder) > 0 Then
        If FolderExists(mFolder) Then BridgeFolder = mFolder: Exit Function
    End If
    mFolder = vbNullString
    Dim cands(0 To 4) As String, i As Long, docs As String
    If Len(FOLDER_OVERRIDE) > 0 Then cands(0) = FOLDER_OVERRIDE
    On Error Resume Next
    docs = CreateObject("WScript.Shell").SpecialFolders("MyDocuments")
    On Error GoTo 0
    If Len(docs) > 0 Then cands(1) = docs & "\FreightSignal"
    cands(2) = Environ$("USERPROFILE") & "\Documents\FreightSignal"
    cands(3) = Environ$("OneDrive") & "\Documents\FreightSignal"
    cands(4) = Environ$("OneDriveCommercial") & "\Documents\FreightSignal"
    For i = 0 To 4
        If Len(cands(i)) > 0 Then
            If FolderExists(cands(i)) Then mFolder = cands(i): Exit For
        End If
    Next i
    BridgeFolder = mFolder
End Function

Private Function FolderExists(ByVal p As String) As Boolean
    On Error Resume Next
    FolderExists = (Len(Dir$(p, vbDirectory)) > 0)
    On Error GoTo 0
End Function

Private Function FeedPath(ByVal fileName As String) As String
    Dim f As String: f = BridgeFolder()
    If Len(f) > 0 Then FeedPath = f & "\" & fileName
End Function

Private Function FileLine(ByVal fileName As String) As String
    Dim p As String: p = FeedPath(fileName)
    If Len(p) > 0 And Len(Dir$(p)) > 0 Then
        FileLine = "  OK       " & fileName & "  (" & Format$(FileDateTime(p), "yyyy-mm-dd hh:nn:ss") & ", " & FileLen(p) & " bytes)" & vbCrLf
    Else
        FileLine = "  MISSING  " & fileName & vbCrLf
    End If
End Function

Private Function ReadAll(ByVal path As String) As String
    On Error GoTo EH
    Dim f As Integer, n As Long, s As String
    If Len(Dir$(path)) = 0 Then Exit Function
    f = FreeFile
    Open path For Binary Access Read Shared As #f
    n = LOF(f)
    If n > 0 Then s = Space$(n): Get #f, , s
    Close #f
    ReadAll = s
    Exit Function
EH:
    On Error Resume Next: Close #f
    LogError "ReadAll " & path
End Function

    Exit Sub

End Sub

Private Function EncodeField(ByVal s As String) As String
    Dim i As Long, c As String, out As String, code As Long
    s = Replace(s, "\", "\\")
    s = Replace(s, vbTab, "\t")
    s = Replace(s, vbCrLf, "\n"): s = Replace(s, vbCr, "\n"): s = Replace(s, vbLf, "\n")
    For i = 1 To Len(s)
        c = Mid$(s, i, 1): code = AscW(c)
        If code >= 32 And code <= 126 Then out = out & c
    Next i
    EncodeField = out
End Function

Private Function DecodeField(ByVal s As String) As String
    Dim i As Long, c As String, n As String, out As String
    If InStr(s, "\") = 0 Then DecodeField = s: Exit Function
    i = 1
    Do While i <= Len(s)
        c = Mid$(s, i, 1)
        If c = "\" And i < Len(s) Then
            i = i + 1: n = Mid$(s, i, 1)
            If n = "t" Then
                out = out & vbTab
            ElseIf n = "n" Then
                out = out & vbCrLf
            Else
                out = out & n
            End If
        Else
            out = out & c
        End If
        i = i + 1
    Loop
    DecodeField = out
End Function

'============================== POSTINGS FEED ==============================
Private Function LoadPostings(ByVal force As Boolean) As Boolean
    On Error GoTo EH
    Dim p As String, stamp As String, txt As String, lines() As String, i As Long, parts() As String, j As Long, hdr() As String
    p = FeedPath(FILE_POSTINGS)
    If Len(p) = 0 Then Exit Function
    If Len(Dir$(p)) = 0 Then Exit Function
    stamp = Format$(FileDateTime(p), "yyyymmddhhnnss") & "|" & FileLen(p)
    If Not force And mPostingsLoaded Then
        If stamp = mPostingsStamp Then LoadPostings = True: Exit Function
        If DateDiff("s", mPostingsLoadedAt, Now) < 2 Then LoadPostings = True: Exit Function
    End If
    txt = ReadAll(p)
    If Len(txt) = 0 Then Exit Function
    txt = Replace(txt, vbCrLf, vbLf)
    lines = Split(txt, vbLf)
    If UBound(lines) < 1 Then Exit Function
    If Left$(lines(0), 22) <> "#FREIGHT-SIGNAL-BRIDGE" Then Exit Function

    Set mCols = CreateObject("Scripting.Dictionary"): mCols.CompareMode = vbTextCompare
    hdr = Split(lines(1), vbTab)
    For j = 0 To UBound(hdr): mCols(LCase$(Trim$(hdr(j)))) = j: Next j
    If Not mCols.Exists("origincity") Or Not mCols.Exists("destcity") Or Not mCols.Exists("postingid") Then Exit Function

    Set mPostings = New Collection
    For i = 2 To UBound(lines)
        If Len(lines(i)) > 0 Then
            parts = Split(lines(i), vbTab)
            If UBound(parts) >= mCols("destcity") Then
                For j = 0 To UBound(parts): parts(j) = DecodeField(parts(j)): Next j
                mPostings.Add parts
            End If
        End If
    Next i
    mPostingsStamp = stamp: mPostingsLoadedAt = Now: mPostingsLoaded = True
    LoadPostings = True
    Exit Function
EH:
    mPostingsLoaded = False
    LogError "LoadPostings"
End Function

Private Function PostingCount() As Long
    If Not mPostings Is Nothing Then PostingCount = mPostings.Count
End Function

' Field of posting #idx (1-based) by lower-case column name.
Private Function PF(ByVal idx As Long, ByVal colName As String) As String
    Dim arr As Variant, c As Long
    If mPostings Is Nothing Or mCols Is Nothing Then Exit Function
    If idx < 1 Or idx > mPostings.Count Then Exit Function
    If Not mCols.Exists(colName) Then Exit Function
    arr = mPostings(idx): c = CLng(mCols(colName))
    If c <= UBound(arr) Then PF = Trim$(CStr(arr(c)))
End Function

Private Function FindPostingIndex(ByVal postingId As String) As Long
    Dim i As Long
    If Len(postingId) = 0 Then Exit Function
    For i = 1 To PostingCount()
        If StrComp(PF(i, "postingid"), postingId, vbTextCompare) = 0 Then FindPostingIndex = i: Exit Function
    Next i
End Function

Private Function PostingTarget(ByVal postingId As String) As Double
    Dim i As Long: i = FindPostingIndex(postingId)

End Function

Private Function BuildLoadDetails(ByVal r As Long) As String

'============================== MATCHING (same scoring as v3) ==============================
Private Function FindMatchingPosting(ByVal subjectText As String, ByVal bodyText As String) As Long
    Dim subj As String, bod As String, i As Long, score As Long, bestScore As Long
    subj = NormalizeForMatching(subjectText): bod = NormalizeForMatching(bodyText): bestScore = -1: mLastScore = 0
    For i = PostingCount() To 1 Step -1                       'bottom-up, strict ">" : the lowest row wins ties (as in v3)
        If Len(PF(i, "origincity")) > 0 And Len(PF(i, "destcity")) > 0 Then
            score = GetLaneMatchScore(subj, PF(i, "origincity"), PF(i, "originstate"), PF(i, "destcity"), PF(i, "deststate"))
            If score >= 0 Then
                score = score + 1000
            Else
                score = GetLaneMatchScore(bod, PF(i, "origincity"), PF(i, "originstate"), PF(i, "destcity"), PF(i, "deststate"))
            End If
            If score > bestScore Then bestScore = score: FindMatchingPosting = i: mLastScore = score
        End If
    Next i
End Function

Private Function GetLaneMatchScore(ByVal text As String, ByVal oc As String, ByVal os As String, ByVal dc As String, ByVal ds As String) As Long
    GetLaneMatchScore = -1
    If Len(text) = 0 Or Not ContainsExactPhrase(text, oc) Or Not ContainsExactPhrase(text, dc) Then Exit Function
    GetLaneMatchScore = 200
    If ContainsExactPhrase(text, os) Then GetLaneMatchScore = GetLaneMatchScore + 10
    If ContainsExactPhrase(text, ds) Then GetLaneMatchScore = GetLaneMatchScore + 10
    If ContainsCityStatePair(text, oc, os) Then GetLaneMatchScore = GetLaneMatchScore + 25
    If ContainsCityStatePair(text, dc, ds) Then GetLaneMatchScore = GetLaneMatchScore + 25
End Function

Private Function ContainsExactPhrase(ByVal text As String, ByVal phrase As String) As Boolean
    phrase = NormalizeForMatching(phrase)
    If Len(phrase) > 0 Then ContainsExactPhrase = (InStr(1, " " & text & " ", " " & phrase & " ", vbTextCompare) > 0)
End Function

Private Function ContainsCityStatePair(ByVal text As String, ByVal city As String, ByVal stateCode As String) As Boolean
    city = NormalizeForMatching(city): stateCode = NormalizeForMatching(stateCode)
    If Len(city) > 0 And Len(stateCode) > 0 Then ContainsCityStatePair = (InStr(1, " " & text & " ", " " & city & " " & stateCode & " ", vbTextCompare) > 0)
End Function

Private Function NormalizeForMatching(ByVal s As String) As String
    Dim chars As Variant, v As Variant
    s = LCase$(CStr(s))
    chars = Array(vbCr, vbLf, vbTab, Chr$(160), ",", ".", "-", "/", "\", ":", ";", "(", ")", "[", "]", "{", "}", "_", "?", "!", "|", ChrW$(8211), ChrW$(8212))
    For Each v In chars: s = Replace(s, CStr(v), " "): Next v
    Do While InStr(s, "  ") > 0: s = Replace(s, "  ", " "): Loop
    ' v6.1: St./Saint, Ft./Fort, Mt./Mount are the same city (the app does the same in js/core/util.js)
    s = " " & s & " "
    s = Replace(s, " saint ", " st "): s = Replace(s, " fort ", " ft "): s = Replace(s, " mount ", " mt ")
    NormalizeForMatching = Trim$(s)
End Function

    End If
End Function

Private Sub RecordEvent(ByVal mail As Outlook.MailItem, ByVal convId As String, ByVal postingId As String, ByVal postingRow As Long, _
                        ByVal stage As String, ByVal carrierRate As String, ByVal targetRate As Double, _
                        ByVal actionText As String, ByVal flagText As String, ByVal eventCount As Long, _
                        Optional ByVal intentText As String = "", Optional ByVal ourRate As Double = 0, Optional ByVal ladStage As String = "", Optional ByVal wasSent As Boolean = False, _
                        Optional ByVal modeText As String = "", Optional ByVal whyText As String = "", Optional ByVal freshText As String = "", Optional ByVal replyText As String = "")
    On Error GoTo EH
    Dim p As String, ln As String, prevRate As Double
    p = FeedPath(FILE_INQUIRIES)
    If Len(p) = 0 Then Exit Sub
    EnsureInquiryFile
    ln = EncodeField(convId) & vbTab & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbTab & EncodeField(GetSenderSMTPAddress(mail)) & vbTab & _

'============================== OUTBOX: drafts made in Freight Signal -> Outlook Drafts ==============================
' Freight Signal lists the drafts it wants in freight-signal-outbox.tsv. For each one not yet in freight-signal-drafts-done.tsv this
' creates an unsent mail item and SAVES it (it lands in Drafts). Nothing is displayed and nothing is sent.
Private Function ProcessOutbox(Optional ByVal force As Boolean = False) As Long
    On Error GoTo EH
    Dim txt As String, lines() As String, i As Long, f() As String, done As Object, id As String, errText As String, made As Long, n As Long
    Dim wantSend As Boolean, sentNow As Boolean, kindText As String
    If mOutboxBusy Then Exit Function
    If Not force Then If DateDiff("s", mOutboxLast, Now) < 3 Then Exit Function
    If Len(BridgeFolder()) = 0 Then Exit Function
    If Len(Dir$(FeedPath(FILE_OUTBOX))) = 0 Then Exit Function
    mOutboxBusy = True: mOutboxLast = Now
    txt = ReadAll(FeedPath(FILE_OUTBOX))
    If Len(txt) = 0 Then GoTo Done
    txt = Replace(txt, vbCrLf, vbLf)
    lines = Split(txt, vbLf)
    ' The app writes "#END<TAB>count" as the very last line. No #END = the file is still being written: try again next time.
    n = UBound(lines)
    Do While n >= 0
        If Len(Trim$(lines(n))) > 0 Then Exit Do
        n = n - 1
    Loop
    If n < 2 Then GoTo Done
    If Left$(lines(n), 4) <> "#END" Then GoTo Done
    Set done = LoadDraftsDone()
    EnsureDraftsDoneFile
    For i = 2 To n - 1
        If Len(lines(i)) > 0 And Left$(lines(i), 1) <> "#" Then
            f = Split(lines(i), vbTab)
            If UBound(f) >= 4 Then
                id = Trim$(DecodeField(f(0)))
                If Len(id) > 0 Then
                    If Not done.Exists(LCase$(id)) Then
                        errText = vbNullString
                        wantSend = False: kindText = "": sentNow = False
                        If UBound(f) >= 5 Then kindText = DecodeField(f(5))
                        If UBound(f) >= 7 Then wantSend = (UCase$(Trim$(f(7))) = "SEND")
                        If wantSend Then
                            If kindText = "Inquiry reply" Or kindText = "Rate response" Or kindText = "Carrier reply" Then wantSend = (mAutoIn And Not IsPaused()) Else wantSend = (mAutoOut And Not IsPaused())
                        End If
                        If CreateOutlookDraft(Trim$(DecodeField(f(1))), Trim$(DecodeField(f(2))), DecodeField(f(3)), DecodeField(f(4)), errText, wantSend, sentNow) Then
                            AppendLine FeedPath(FILE_DRAFTS_DONE), id & vbTab & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbTab & IIf(sentNow, "SENT", "OK") & vbTab & ""
                            made = made + 1
                        Else
                            AppendLine FeedPath(FILE_DRAFTS_DONE), id & vbTab & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbTab & "ERROR" & vbTab & EncodeField(errText)
                        End If
                        done(LCase$(id)) = True
                    End If
                End If
            End If
        End If
    Next i
    If made > 0 Then
        mDraftsMade = mDraftsMade + made
        WriteStatus made & " draft(s) saved to Outlook Drafts"
    End If
Done:
    mOutboxBusy = False
    ProcessOutbox = made
    Exit Function
EH:
    mOutboxBusy = False
    LogError "ProcessOutbox"
End Function

Private Function CreateOutlookDraft(ByVal toAddr As String, ByVal ccAddr As String, ByVal subj As String, ByVal body As String, ByRef errText As String, _
                                   Optional ByVal wantSend As Boolean = False, Optional ByRef wasSent As Boolean = False) As Boolean
    On Error GoTo EH
    Dim m As Outlook.MailItem
    wasSent = False
    Set m = Application.CreateItem(0)          '0 = olMailItem
    m.To = toAddr
    If Len(ccAddr) > 0 Then m.CC = ccAddr
    m.Subject = subj
    m.Body = body
    If MaySend(wantSend, True, toAddr, 1) And InStr(toAddr, ";") = 0 And InStr(toAddr, ",") = 0 Then
        m.Send: wasSent = True: NoteSend "outbox", "outreach"          'ONLY when the Danger-zone switch is on and every guard in MaySend agrees
    Else
        m.Save                                      'unsent + saved = it sits in Drafts.
    End If
    CreateOutlookDraft = True
    Exit Function
EH:
    errText = Err.Description
    LogError "CreateOutlookDraft"
End Function

Private Function LoadDraftsDone() As Object
    Dim d As Object, txt As String, lines() As String, i As Long, c() As String
    Set d = CreateObject("Scripting.Dictionary"): d.CompareMode = vbTextCompare
    txt = ReadAll(FeedPath(FILE_DRAFTS_DONE))
    If Len(txt) > 0 Then
        txt = Replace(txt, vbCrLf, vbLf)
        lines = Split(txt, vbLf)
        For i = 0 To UBound(lines)
            If Len(lines(i)) > 0 And Left$(lines(i), 7) <> "DraftId" And Left$(lines(i), 1) <> "#" Then
                c = Split(lines(i), vbTab)
                If UBound(c) >= 0 Then d(LCase$(Trim$(DecodeField(c(0))))) = True
            End If
        Next i
    End If
    Set LoadDraftsDone = d
End Function

Private Function GetSenderSMTPAddress(ByVal mail As Outlook.MailItem) As String
    On Error GoTo Fallback
    Dim exUser As Object
    If UCase$(mail.SenderEmailType) = "EX" Then
        Set exUser = mail.Sender.GetExchangeUser
        If Not exUser Is Nothing Then GetSenderSMTPAddress = exUser.PrimarySmtpAddress: Exit Function
    End If

    On Error Resume Next
    GetSenderSMTPAddress = mail.SenderEmailAddress
End Function

Private Function IsBlockedSender(ByVal mail As Outlook.MailItem) As Boolean

    s = LCase$(GetSenderSMTPAddress(mail))
    For Each v In mBlock
        If InStr(1, s, CStr(v), vbTextCompare) > 0 Then IsBlockedSender = True: Exit Function
    Next v
End Function

                          InStr(subj, "out of office") > 0 Or InStr(subj, "out-of-office") > 0 Or InStr(subj, "automatic reply") > 0 Or _
                          InStr(subj, "auto-reply") > 0 Or InStr(subj, "autoreply") > 0 Or InStr(subj, "undeliverable") > 0 Or InStr(subj, "delivery status notification") > 0)

    Dim a As TAnalysis, bad As Boolean
    AnalyseEmail body, mSelfOwn, a
    mSelfN = mSelfN + 1
    bad = (a.Intent <> expectIntent)
    If expectRate > 0 And a.Rate <> expectRate Then bad = True
    If expectConf = "weak" And Not a.Weak Then bad = True
    If expectConf = "strong" And (a.Weak Or Not a.Found) Then bad = True
    If bad Then
        mSelfFail = mSelfFail + 1
        If mSelfFail <= 12 Then mSelfMsg = mSelfMsg & nm & ": expected " & expectIntent & IIf(expectRate > 0, " " & expectRate, "") & ", got " & a.Intent & IIf(a.Found, " " & a.Rate, "") & IIf(a.Weak, " (weak)", "") & vbCrLf
    End If
End Sub

Private Function SafeNumeric(ByVal value As Variant) As Double
    If IsNumeric(value) Then SafeNumeric = CDbl(value)
End Function

Private Sub LogError(ByVal procedureName As String)
    Dim msg As String
    msg = Format$(Now, "yyyy-mm-dd hh:nn:ss") & " " & procedureName & " error " & Err.Number & ": " & Err.Description
    Debug.Print msg
    On Error Resume Next
    If Len(BridgeFolder()) > 0 Then AppendLine FeedPath(FILE_ERRORS), msg
End Sub
